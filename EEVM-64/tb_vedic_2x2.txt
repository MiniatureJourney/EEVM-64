// =============================================================================
// Testbench : tb_vedic_2x2
// DUT       : vedic_2x2
// Spec ref  : EEVM-64_architecturePlan.pdf §4 verification hooks -
//             "vedic_2x2: exhaustive (all input combinations)"
//
// Method    : Exhaustive sweep of all 16 (a,b) combinations. Self-checking
//             against a behavioral golden model (plain unsigned '*'), which
//             is the reference model EEVM-64.pdf §6.2 specifies for
//             functional verification. Dumps a VCD so the same waveform can
//             be opened in GTKWave or Vivado's simulator.
//
// Pass/fail : Prints a PASS/FAIL summary line and a nonzero error count if
//             any vector mismatches. This is the line to paste into the
//             testingDataBlockwise doc for this block.
// =============================================================================
`timescale 1ns / 1ps

module tb_vedic_2x2;

    logic [1:0] a, b;
    logic [3:0] p;
    logic [3:0] expected;
    integer errors;
    integer i, j;

    // ---- DUT instantiation ---------------------------------------------------
    vedic_2x2 dut (
        .a(a),
        .b(b),
        .p(p)
    );

    // ---- Stimulus + self-checking ---------------------------------------------
    initial begin
        $dumpfile("tb_vedic_2x2.vcd");
        $dumpvars(0, tb_vedic_2x2);

        errors = 0;

        for (i = 0; i < 4; i = i + 1) begin
            for (j = 0; j < 4; j = j + 1) begin
                a = i[1:0];
                b = j[1:0];
                #5;                       // let combinational logic settle
                expected = i * j;         // golden reference model (unsigned)

                if (p !== expected) begin
                    errors = errors + 1;
                    $display("FAIL: a=%0d b=%0d -> p=%0d (expected %0d)",
                              a, b, p, expected);
                end
            end
        end

        #5;
        if (errors == 0)
            $display("PASS: vedic_2x2 exhaustive verification - 16/16 vectors correct.");
        else
            $display("FAIL: vedic_2x2 - %0d/16 vectors mismatched.", errors);

        $finish;
    end

endmodule
